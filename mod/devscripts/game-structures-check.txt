# #181: with no blueprint files of its own, the game copies its village houses, shipwrecks and ruins into
# config/kindredspirit/blueprints-from-minecraft at startup, describes them, and she builds two by name.
world kindred-game-structures
wait 220
cmd gamemode creative
cmd time set 1000
cmd weather clear
cmd gamerule doMobSpawning false
cmd gamerule doDaylightCycle false
cmd kill @e[type=!minecraft:player]
cmd tp @s 0.5 200 0.5 0 10
wait 40
cmd fill -24 194 -24 24 198 40 minecraft:stone
cmd fill -24 199 -24 24 199 40 minecraft:grass_block
cmd fill -24 200 -24 24 207 40 minecraft:air
cmd fill -24 208 -24 24 216 40 minecraft:air
type /kindred style magic
wait 20
cmd tp @s 0.5 200 0.5 0 35
wait 10
do blueprint {"what": "village_taiga_medium_house_1"}
wait 400
cmd tp @s 0.5 207 -8.5 0 25
wait 20
screenshot game-structures-taiga-house
cmd tp @s 0.5 200 18.5 0 35
wait 10
do blueprint {"what": "igloo"}
wait 160
cmd tp @s 0.5 206 10.5 0 30
wait 20
screenshot game-structures-igloo
quit
